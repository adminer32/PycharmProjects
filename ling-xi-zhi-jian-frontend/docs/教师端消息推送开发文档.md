# 教师端消息推送功能开发文档

## 概述

学生端已实现 WebSocket 实时接收教师提醒消息的功能。教师端需要配合修改，以便向指定学生或全体学生发送提醒消息。

---

## 一、学生端实现概况

### 1. WebSocket 端点
```
学生端连接地址: /ws/teacher/notification?token={学生token}
```

### 2. 学生端功能特性
- **自动连接**: 页面加载时自动建立 WebSocket 连接
- **心跳检测**: 每 30 秒发送一次 ping，保持连接活跃
- **自动重连**: 断开后自动重连，最多 10 次
- **消息弹窗**: 使用右下角浮动通知组件显示消息
- **消息存储**: 在 store 中保留最近的消息记录

### 3. 接收消息格式
```typescript
interface TeacherNotification {
    id: string;                    // 消息唯一标识
    type: 'info' | 'warning' | 'success' | 'error';  // 消息类型
    title: string;                 // 消息标题
    message: string;               // 消息内容
    timestamp: number;             // 时间戳
}
```

### 4. 学生端代码位置
- Store: `src/stores/teacherNotificationStore.ts`
- 初始化: `src/App.vue` (onMounted 中调用 initWebSocket)

## 二、教师端后端改造

### 1. WebSocket 连接管理

需要维护一个学生用户 ID 到 WebSocket 连接的映射表：

```javascript
// 示例：维护连接映射
const studentConnections = new Map(); // userId -> WebSocket

// 或使用 Set 存储所有连接（用于广播）
const allConnections = new Set();
```

### 2. 学生端连接处理

当学生端连接 `/ws/teacher/notification` 时：

```javascript
// 1. 解析 token 获取用户 ID
function getUserIdFromToken(token) {
    // 根据你们的 token 机制解析用户 ID
    // 返回 userId
}

// 2. WebSocket 连接建立时
wss.on('connection', (ws, req) => {
    const url = new URL(req.url, `http://${req.headers.host}`);
    const token = url.searchParams.get('token');
    const userId = getUserIdFromToken(token);

    // 保存连接
    studentConnections.set(userId, ws);
    allConnections.add(ws);

    // 心跳/保活处理
    ws.isAlive = true;
    ws.on('pong', () => { ws.isAlive = true; });

    // 断开时移除
    ws.on('close', () => {
        studentConnections.delete(userId);
        allConnections.delete(ws);
    });
});
```

### 3. 心跳检测（可选但推荐）

```javascript
// 定期检测连接是否存活
const heartbeatInterval = setInterval(() => {
    wss.clients.forEach((ws) => {
        if (ws.isAlive === false) {
            return ws.terminate();
        }
        ws.isAlive = false;
        ws.ping();
    });
}, 30000);
```

### 4. 消息发送机制

#### 4.1 发送给指定学生
```javascript
function sendToStudent(userId, message) {
    const ws = studentConnections.get(userId);
    if (ws && ws.readyState === WebSocket.OPEN) {
        ws.send(JSON.stringify({
            id: generateUUID(),
            type: message.type || 'info',
            title: message.title || '教师提醒',
            message: message.content,
            timestamp: Date.now()
        }));
        return true;
    }
    return false;
}
```

#### 4.2 广播给所有学生
```javascript
function broadcastToAll(message) {
    const payload = JSON.stringify({
        id: generateUUID(),
        type: message.type || 'info',
        title: message.title || '教师提醒',
        message: message.content,
        timestamp: Date.now()
    });

    let successCount = 0;
    allConnections.forEach((ws) => {
        if (ws.readyState === WebSocket.OPEN) {
            ws.send(payload);
            successCount++;
        }
    });
    return successCount;
}
```

### 5. 教师端调用示例

#### 方式一：通过 HTTP API 触发
```javascript
// 教师端前端调用
async function sendNotification(targetUserId, message) {
    await fetch('/api/teacher/notify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            targetUserId,  // 'all' 表示全体，或具体 userId
            type: message.type,
            title: message.title,
            content: message.content
        })
    });
}

// 教师端后端 API
app.post('/api/teacher/notify', async (req, res) => {
    const { targetUserId, type, title, content } = req.body;

    if (targetUserId === 'all') {
        const count = broadcastToAll({ type, title, content });
        res.json({ success: true, delivered: count });
    } else {
        const success = sendToStudent(targetUserId, { type, title, content });
        res.json({ success });
    }
});
```

#### 方式二：教师端直连 WebSocket
```javascript
// 教师端 WebSocket 连接（如果需要教师端也用 WebSocket）

}));
```

## 三、消息格式说明

### 发送方（教师端）消息格式
```json
{
    "targetUserId": "student_123",  // 必填：目标用户ID，或 "all" 表示全体
    "type": "info",                 // 必填：消息类型
    "title": "上课提醒",            // 可选：消息标题
```

### 接收方（学生端）消息格式
```json
{
    "id": "uuid-xxx",
    "type": "info",
    "title": "上课提醒",
    "message": "同学们，下节课5分钟后开始",
```

### type 可选值
| 值 | 含义 | 样式 |
|---|---|---|
| `info` | 普通信息 | 蓝色 |
| `success` | 成功提示 | 绿色 |
| `warning` | 警告提示 | 黄色 |
| `error` | 错误提示 | 红色 |

## 四、完整后端示例（Node.js + ws）

```javascript
const WebSocket = require('ws');
const wss = new WebSocket.Server({ port: 8080 });

// 连接管理
const studentConnections = new Map();

// 解析 token（根据你们实际机制修改）
function getUserIdFromToken(token) {
    // 示例：从 JWT 解析或查询数据库
    // 这里需要你们根据实际 token 机制实现
    return userId;
}

// 生成 UUID
function generateUUID() {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
        const r = Math.random() * 16 | 0;
        const v = c === 'x' ? r : (r & 0x3 | 0x8);
        return v.toString(16);
    });
}

// 广播消息
function broadcast(message) {
    const payload = JSON.stringify({
        id: generateUUID(),
        ...message,
        timestamp: Date.now()
    });

    if (message.targetUserId === 'all') {
        // 发送给所有学生
        studentConnections.forEach((ws) => {
            if (ws.readyState === WebSocket.OPEN) {
                ws.send(payload);
            }
        });
    } else {
        // 发送给指定学生
        const ws = studentConnections.get(message.targetUserId);
        if (ws && ws.readyState === WebSocket.OPEN) {
            ws.send(payload);
        }
    }
}

// WebSocket 连接处理
wss.on('connection', (ws, req) => {
    const url = new URL(req.url, `http://${req.headers.host}`);
    const token = url.searchParams.get('token');

    if (!token) {
        ws.close();
        return;
    }

    const userId = getUserIdFromToken(token);
    if (!userId) {
        ws.close();
        return;
    }

    // 保存连接
    studentConnections.set(userId, ws);
    ws.userId = userId;
    ws.isAlive = true;

    console.log(`学生 ${userId} 已连接`);

    // 心跳
    ws.on('pong', () => { ws.isAlive = true; });

    // 消息处理
    ws.on('message', (data) => {
        try {
            const message = JSON.parse(data);

            // 教师端发送的消息
            if (message.action === 'send') {
                broadcast({
                    type: message.type,
                    title: message.title,
                    message: message.content,
                    targetUserId: message.targetUserId
                });
            }
        } catch (e) {
            console.error('消息解析失败:', e);
        }
    });

    // 断开处理
    ws.on('close', () => {
        if (ws.userId) {
            studentConnections.delete(ws.userId);
            console.log(`学生 ${ws.userId} 已断开`);
        }
    });

    ws.on('error', (err) => {
        console.error('WebSocket 错误:', err);
    });
});

// 心跳检测
const heartbeat = setInterval(() => {
    wss.clients.forEach((ws) => {
        if (ws.isAlive === false) {
            return ws.terminate();
        }
        ws.isAlive = false;
        ws.ping();
    });
}, 30000);

wss.on('close', () => {
    clearInterval(heartbeat);
});
```

## 五、注意事项

### 1. Token 安全
- WebSocket 连接时传递 token 进行身份验证
- 确保 token 验证机制安全可靠
- 防止未授权用户连接到 WebSocket 服务

### 2. 连接数限制
- 根据服务器性能设置最大连接数
- 做好连接超时和空闲连接清理

### 3. 消息大小
- 建议单条消息不超过 1KB
- 避免发送过大的数据
